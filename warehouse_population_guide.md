# Supabase DW Population Guide (From `stg_*` Tables)

This script populates your warehouse tables from the staging tables you already loaded.

Assumptions:
- You are using PostgreSQL/Supabase.
- All `stg_*` tables and all `dim_*` / `fact_*` tables already exist.
- `fact_reviews` should have one row per `review_id` (so if an order has multiple items, we map to one representative item: the smallest `order_item_id`).

## 2) Populate `dim_date`

This fills dates from the minimum to maximum date found in staging timestamps.

```sql
WITH raw_dates AS (
    SELECT order_purchase_timestamp::date AS d FROM stg_orders WHERE order_purchase_timestamp IS NOT NULL
    UNION
    SELECT order_approved_at::date FROM stg_orders WHERE order_approved_at IS NOT NULL
    UNION
    SELECT order_delivered_carrier_date::date FROM stg_orders WHERE order_delivered_carrier_date IS NOT NULL
    UNION
    SELECT order_delivered_customer_date::date FROM stg_orders WHERE order_delivered_customer_date IS NOT NULL
    UNION
    SELECT order_estimated_delivery_date::date FROM stg_orders WHERE order_estimated_delivery_date IS NOT NULL
    UNION
    SELECT review_creation_date::date FROM stg_order_reviews WHERE review_creation_date IS NOT NULL
    UNION
    SELECT review_answer_timestamp::date FROM stg_order_reviews WHERE review_answer_timestamp IS NOT NULL
    UNION
    SELECT shipping_limit_date::date FROM stg_order_items WHERE shipping_limit_date IS NOT NULL
), bounds AS (
    SELECT MIN(d) AS min_d, MAX(d) AS max_d
    FROM raw_dates
), date_series AS (
    SELECT gs::date AS full_date
    FROM bounds,
         generate_series(bounds.min_d, bounds.max_d, interval '1 day') gs
)
INSERT INTO dim_date (
    date_key,
    full_date,
    day_num,
    month_num,
    month_name,
    quarter_num,
    year_num,
    week_of_year,
    is_weekend,
    is_holiday
)
SELECT
    to_char(full_date, 'YYYYMMDD')::int AS date_key,
    full_date,
    EXTRACT(day FROM full_date)::int,
    EXTRACT(month FROM full_date)::int,
    to_char(full_date, 'Month')::varchar(20),
    EXTRACT(quarter FROM full_date)::int,
    EXTRACT(year FROM full_date)::int,
    EXTRACT(week FROM full_date)::int,
    CASE WHEN EXTRACT(isodow FROM full_date) IN (6,7) THEN 'Y' ELSE 'N' END AS is_weekend,
    'N' AS is_holiday
FROM date_series
ON CONFLICT (date_key) DO NOTHING;
```

## 6) Populate `dim_geography`

`dim_geography` has no unique constraint in your DDL, so for repeatable loads, full refresh is recommended.

```sql
WITH geo_base AS (
    SELECT
        g.geolocation_zip_code_prefix::text AS zip_prefix,
        g.geolocation_city AS city,
        g.geolocation_state AS state_code,
        AVG(g.geolocation_lat)::numeric(10,6) AS latitude,
        AVG(g.geolocation_lng)::numeric(10,6) AS longitude
    FROM stg_geolocation g
    GROUP BY
        g.geolocation_zip_code_prefix,
        g.geolocation_city,
        g.geolocation_state
)
INSERT INTO dim_geography (
    zip_prefix,
    city,
    state_code,
    latitude,
    longitude,
    region_group
)
SELECT
    gb.zip_prefix,
    gb.city,
    gb.state_code,
    gb.latitude,
    gb.longitude,
    CASE
        WHEN gb.state_code IN ('AC','AP','AM','PA','RO','RR','TO') THEN 'North'
        WHEN gb.state_code IN ('AL','BA','CE','MA','PB','PE','PI','RN','SE') THEN 'Northeast'
        WHEN gb.state_code IN ('DF','GO','MT','MS') THEN 'Midwest'
        WHEN gb.state_code IN ('ES','MG','RJ','SP') THEN 'Southeast'
        WHEN gb.state_code IN ('PR','RS','SC') THEN 'South'
        ELSE 'Unknown'
    END AS region_group
FROM geo_base gb;
```

## 9) Populate `fact_sales`

```sql
INSERT INTO fact_sales (
    order_id,
    order_item_id,
    date_key_purchase,
    date_key_delivered,
    customer_key,
    seller_key,
    product_key,
    geo_customer_key,
    geo_seller_key,
    order_status_key,
    item_price,
    freight_value,
    gross_item_value,
    item_count,
    delivery_days,
    delay_days
)
SELECT
    oi.order_id,
    oi.order_item_id,
    to_char(o.order_purchase_timestamp::date, 'YYYYMMDD')::int AS date_key_purchase,
    CASE
        WHEN o.order_delivered_customer_date IS NOT NULL
        THEN to_char(o.order_delivered_customer_date::date, 'YYYYMMDD')::int
        ELSE NULL
    END AS date_key_delivered,
    dc.customer_key,
    ds.seller_key,
    dp.product_key,
    dgc.geo_key AS geo_customer_key,
    dgs.geo_key AS geo_seller_key,
    dos.order_status_key,
    oi.price::numeric(12,2) AS item_price,
    oi.freight_value::numeric(12,2) AS freight_value,
    (oi.price + oi.freight_value)::numeric(12,2) AS gross_item_value,
    1 AS item_count,
    CASE
        WHEN o.order_delivered_customer_date IS NOT NULL
         AND o.order_purchase_timestamp IS NOT NULL
        THEN EXTRACT(EPOCH FROM (o.order_delivered_customer_date - o.order_purchase_timestamp)) / 86400.0
        ELSE NULL
    END::numeric(6,2) AS delivery_days,
    CASE
        WHEN o.order_delivered_customer_date IS NOT NULL
         AND o.order_estimated_delivery_date IS NOT NULL
        THEN GREATEST(
            -- subtraction of two dates returns integer days, so cast to numeric
            (o.order_delivered_customer_date::date - o.order_estimated_delivery_date::date)::numeric,
            0
        )
        ELSE NULL
    END::numeric(6,2) AS delay_days
FROM stg_order_items oi
JOIN stg_orders o
    ON o.order_id = oi.order_id
LEFT JOIN dim_customer dc
    ON dc.customer_id = o.customer_id
   AND dc.is_current = 'Y'
LEFT JOIN dim_seller ds
    ON ds.seller_id = oi.seller_id
LEFT JOIN dim_product dp
    ON dp.product_id = oi.product_id
LEFT JOIN stg_customers sc
    ON sc.customer_id = o.customer_id
LEFT JOIN dim_geography dgc
    ON dgc.zip_prefix = sc.customer_zip_code_prefix::text
   AND dgc.city = sc.customer_city
   AND dgc.state_code = sc.customer_state
LEFT JOIN stg_sellers ss
    ON ss.seller_id = oi.seller_id
LEFT JOIN dim_geography dgs
    ON dgs.zip_prefix = ss.seller_zip_code_prefix::text
   AND dgs.city = ss.seller_city
   AND dgs.state_code = ss.seller_state
LEFT JOIN dim_order_status dos
    ON dos.order_status = o.order_status
ON CONFLICT (order_id, order_item_id) DO UPDATE
SET
    date_key_purchase = EXCLUDED.date_key_purchase,
    date_key_delivered = EXCLUDED.date_key_delivered,
    customer_key = EXCLUDED.customer_key,
    seller_key = EXCLUDED.seller_key,
    product_key = EXCLUDED.product_key,
    geo_customer_key = EXCLUDED.geo_customer_key,
    geo_seller_key = EXCLUDED.geo_seller_key,
    order_status_key = EXCLUDED.order_status_key,
    item_price = EXCLUDED.item_price,
    freight_value = EXCLUDED.freight_value,
    gross_item_value = EXCLUDED.gross_item_value,
    item_count = EXCLUDED.item_count,
    delivery_days = EXCLUDED.delivery_days,
    delay_days = EXCLUDED.delay_days;
```

## 11) Populate `fact_reviews`

Because `fact_reviews` is unique on `review_id`, this maps each review to a single product/seller using the first order item.

```sql
WITH first_item_per_order AS (
    SELECT DISTINCT ON (oi.order_id)
        oi.order_id,
        oi.product_id,
        oi.seller_id
    FROM stg_order_items oi
    ORDER BY oi.order_id, oi.order_item_id
)
INSERT INTO fact_reviews (
    review_id,
    order_id,
    date_key_review,
    customer_key,
    seller_key,
    product_key,
    order_status_key,
    review_score,
    review_count,
    response_time_hours
)
SELECT
    r.review_id,
    r.order_id,
    CASE
        WHEN r.review_creation_date IS NOT NULL
        THEN to_char(r.review_creation_date::date, 'YYYYMMDD')::int
        ELSE NULL
    END AS date_key_review,
    dc.customer_key,
    ds.seller_key,
    dp.product_key,
    dos.order_status_key,
    r.review_score,
    1 AS review_count,
    CASE
        WHEN r.review_answer_timestamp IS NOT NULL
         AND r.review_creation_date IS NOT NULL
        THEN EXTRACT(EPOCH FROM (r.review_answer_timestamp - r.review_creation_date)) / 3600.0
        ELSE NULL
    END::numeric(10,2) AS response_time_hours
FROM stg_order_reviews r
JOIN stg_orders o
    ON o.order_id = r.order_id
LEFT JOIN first_item_per_order fi
    ON fi.order_id = r.order_id
LEFT JOIN dim_customer dc
    ON dc.customer_id = o.customer_id
   AND dc.is_current = 'Y'
LEFT JOIN dim_seller ds
    ON ds.seller_id = fi.seller_id
LEFT JOIN dim_product dp
    ON dp.product_id = fi.product_id
LEFT JOIN dim_order_status dos
    ON dos.order_status = o.order_status
ON CONFLICT (review_id) DO UPDATE
SET
    order_id = EXCLUDED.order_id,
    date_key_review = EXCLUDED.date_key_review,
    customer_key = EXCLUDED.customer_key,
    seller_key = EXCLUDED.seller_key,
    product_key = EXCLUDED.product_key,
    order_status_key = EXCLUDED.order_status_key,
    review_score = EXCLUDED.review_score,
    review_count = EXCLUDED.review_count,
    response_time_hours = EXCLUDED.response_time_hours;
```
